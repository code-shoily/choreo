Dataflow & Processing Pipelines
Choreo.Dataflow
alias Choreo.Dataflow

pipeline =
  Dataflow.new()
  |> Dataflow.add_source(:kafka, label: "Kafka Ingest")
  |> Dataflow.add_transform(:parser, label: "JSON Parser")
  |> Dataflow.add_transform(:enricher, label: "Enricher")
  |> Dataflow.add_sink(:postgres, label: "Postgres Sink")
  |> Dataflow.add_sink(:s3, label: "S3 Archive")
  |> Dataflow.connect(:kafka, :parser)
  |> Dataflow.connect(:parser, :enricher)
  |> Dataflow.connect(:enricher, :postgres)
  |> Dataflow.connect(:enricher, :s3)

Visual Architectures of Any Type

Switch tabs below to see how standard Elixir declarations render into clean, readable system layouts.

alias Choreo.FSM

fsm =
  FSM.new()
  |> FSM.add_initial_state(:idle)
  |> FSM.add_state(:running)
  |> FSM.add_final_state(:done)
  |> FSM.add_transition(:idle, :running, label: "start")
  |> FSM.add_transition(:running, :done, label: "finish")
alias Choreo.Sequence

seq =
  Sequence.new()
  |> Sequence.add_actor(:user, label: "User")
  |> Sequence.add_participant(:api, label: "API")
  |> Sequence.add_participant(:db, label: "Database")
  |> Sequence.message(:user, :api, label: "GET /accounts")
  |> Sequence.activate(:api)
  |> Sequence.message(:api, :db, label: "SELECT")
  |> Sequence.return(:db, :api, label: "rows")
  |> Sequence.deactivate(:api)
  |> Sequence.return(:api, :user, label: "200 OK")
alias Choreo.Workflow

workflow =
  Workflow.new()
  |> Workflow.add_start(:received)
  |> Workflow.add_task(:charge, timeout_ms: 5000)
  |> Workflow.add_compensation(:refund, for: :charge)
  |> Workflow.add_end(:done)
  |> Workflow.connect(:received, :charge)
  |> Workflow.connect(:charge, :done)
alias Choreo.Domain

domain =
  Domain.new()
  |> Domain.add_aggregate(:order, label: "Order")
  |> Domain.add_type(:line_item, label: "OrderLine")
  |> Domain.connect(:order, :line_item, label: "has")
